rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() {
      return request.auth != null;
    }

    function isSelf(uid) {
      return signedIn() && request.auth.uid == uid;
    }

    function isAdminEmail() {
      return signedIn() && request.auth.token.email == "lts7364@gmail.com";
    }

    function hasAdminRole() {
      return signedIn()
        && exists(/databases/$(database)/documents/users/$(request.auth.uid))
        && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == "admin";
    }

    function isAdmin() {
      return isAdminEmail() || hasAdminRole();
    }

    function isNotBlocked() {
      return signedIn()
        && exists(/databases/$(database)/documents/users/$(request.auth.uid))
        && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.blocked != true;
    }

    match /users/{uid} {
      allow create: if isSelf(uid)
        && request.resource.data.email == request.auth.token.email
        && request.resource.data.approved == true
        && request.resource.data.blocked == false;

      allow read: if isSelf(uid) || isAdmin();
      allow update: if isAdmin();
      allow delete: if false;
    }

    match /downloadLogs/{logId} {
      allow create: if signedIn() && isNotBlocked();
      allow read: if isAdmin();
      allow update, delete: if false;
    }

    match /requests/{requestId} {
      allow create: if signedIn()
        && isNotBlocked()
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.anonymous == true;

      allow read: if isAdmin() || (signedIn() && resource.data.uid == request.auth.uid);
      allow update: if isAdmin();
      allow delete: if isAdmin() || (signedIn() && resource.data.uid == request.auth.uid && resource.data.status in ["적용완료", "보류", "거절"]);
    }
  }
}
